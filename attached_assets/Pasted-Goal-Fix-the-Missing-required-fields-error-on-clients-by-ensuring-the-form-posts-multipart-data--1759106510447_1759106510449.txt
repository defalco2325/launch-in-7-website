------------------ BEGIN FILE: netlify/functions/client-onboarding.js ------------------
import * as multipart from 'aws-lambda-multipart-parser';
import sgMail from '@sendgrid/mail';

sgMail.setApiKey(process.env.SENDGRID_API_KEY);

const H = (h, k) => h?.[k] ?? h?.[k.toLowerCase()] ?? h?.[k.toUpperCase()];

const val = (v) => {
  if (v == null) return '';
  if (Array.isArray(v)) return v.map(val).filter(Boolean).join(', ');
  if (typeof v === 'object' && 'value' in v) return val(v.value);
  return String(v);
};

const extractFiles = (obj) => {
  const out = [];
  for (const [k, v] of Object.entries(obj || {})) {
    const pushFile = (f) => {
      if (!f) return;
      if (f.content && (Buffer.isBuffer(f.content) || typeof f.content === 'string')) {
        out.push({
          field: k,
          filename: f.filename || 'attachment',
          type: f.contentType || 'application/octet-stream',
          content: Buffer.isBuffer(f.content) ? f.content.toString('base64') : f.content,
        });
      }
    };
    if (Array.isArray(v)) v.forEach(pushFile);
    else if (v && typeof v === 'object' && ('filename' in v || 'content' in v)) pushFile(v);
  }
  return out;
};

export const handler = async (event) => {
  if (event.httpMethod !== 'POST') {
    return { statusCode: 405, body: 'Method Not Allowed' };
  }

  try {
    const rawCT = H(event.headers || {}, 'content-type') || '';
    const ct = rawCT.toLowerCase();

    let parsed = {};
    let files = [];

    if (ct.includes('multipart/form-data')) {
      parsed = multipart.parse(event, true) || {};
      files = extractFiles(parsed);
      // Remove file-like entries before field validation
      for (const k of Object.keys(parsed)) {
        const v = parsed[k];
        if (v && typeof v === 'object' && ('filename' in v || 'content' in v)) {
          delete parsed[k];
        }
      }
    } else if (ct.includes('application/x-www-form-urlencoded')) {
      const sp = new URLSearchParams(event.body || '');
      for (const [k, v] of sp.entries()) parsed[k] = v;
    } else if (ct.includes('application/json')) {
      parsed = JSON.parse(event.body || '{}');
    } else {
      try { parsed = JSON.parse(event.body || '{}'); } catch { parsed = {}; }
    }

    // Normalize required fields
    const businessName = val(parsed.businessName).trim();
    const contactName  = val(parsed.contactName).trim();
    const email        = val(parsed.email).trim();

    // DEBUG (temporary): leave enabled for first submission after deploy
    console.log('CT:', rawCT, 'Keys:', Object.keys(parsed));
    console.log('Required:', { businessName, contactName, email });
    console.log('Files:', files.map(f => ({ field: f.field, filename: f.filename, type: f.type })));

    if (!businessName || !contactName || !email) {
      return {
        statusCode: 400,
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          ok: false,
          error: 'Missing required fields',
          received: {
            businessName: !!businessName,
            contactName:  !!contactName,
            email:        !!email,
          },
          hint: 'Ensure multipart/form-data and exact field names (businessName, contactName, email).',
        }),
      };
    }

    const toEmail = process.env.ONBOARDING_TO_EMAIL || 'team@launchin7.io';
    const fromEmail = process.env.FROM_EMAIL || 'Team <team@launchin7.io>';

    const html = `
      <h2>New Client Onboarding</h2>
      <p><strong>Business:</strong> ${businessName}</p>
      <p><strong>Contact:</strong> ${contactName}</p>
      <p><strong>Email:</strong> ${email}</p>
      <p><strong>Website:</strong> ${val(parsed.website)}</p>
      <p><strong>Phone:</strong> ${val(parsed.phone)}</p>
      <p><strong>Tagline:</strong> ${val(parsed.tagline)}</p>
      <p><strong>Short Description:</strong> ${val(parsed.shortDescription)}</p>
      <p><strong>Pages:</strong> ${val(parsed.pagesSelected)}</p>
      <p><strong>Features:</strong> ${val(parsed.featuresSelected)}</p>
      <p><strong>Copywriting:</strong> ${val(parsed.copywriting)}</p>
      <p><strong>SEO:</strong> ${val(parsed.seo)}</p>
      <p><strong>Timeline:</strong> ${val(parsed.timeline)}</p>
      <p><strong>Package Interest:</strong> ${val(parsed.packageInterest)}</p>
    `.trim();

    const msg = {
      to: toEmail,
      from: fromEmail, // must be verified on SendGrid
      subject: `Onboarding • ${businessName}`,
      html,
      attachments: files.map((f) => ({
        content: f.content,
        filename: f.filename,
        type: f.type,
        disposition: 'attachment',
      })),
    };

    await sgMail.send(msg);

    return {
      statusCode: 200,
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ ok: true, message: 'Submission sent.' }),
    };
  } catch (err) {
    console.error('Handler error:', err);
    return {
      statusCode: 500,
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ ok: false, error: 'Server error while processing form.' }),
    };
  }
};
