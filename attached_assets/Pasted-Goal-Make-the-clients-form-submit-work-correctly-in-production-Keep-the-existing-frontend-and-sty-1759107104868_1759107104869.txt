import * as multipart from 'aws-lambda-multipart-parser';
import sgMail from '@sendgrid/mail';

/**
 * ENV VARS (set in Netlify → Site settings → Build & deploy → Environment):
 * - SENDGRID_API_KEY  (required)
 * - FROM_EMAIL        (e.g., "LaunchIn7 Team <team@launchin7.io>", must be verified in SendGrid)
 * - ONBOARDING_TO_EMAIL (defaults to "team@launchin7.io" if not set)
 */
sgMail.setApiKey(process.env.SENDGRID_API_KEY);

const H = (h, k) => h?.[k] ?? h?.[k.toLowerCase()] ?? h?.[k.toUpperCase()];

// Normalize a field value into a clean string.
const str = (v) => {
  if (v == null) return '';
  if (Array.isArray(v)) return v.map(str).filter(Boolean).join(', ');
  if (typeof v === 'object' && 'value' in v) return str(v.value);
  return String(v);
};

// Extract files from aws-lambda-multipart-parser output.
// Supports single and multiple files on the same field name (logoFiles, photos, otherAssets, brandGuide).
const extractFiles = (obj) => {
  const out = [];
  for (const [field, val] of Object.entries(obj || {})) {
    const pushFile = (f) => {
      if (!f) return;
      const hasContent = f.content && (Buffer.isBuffer(f.content) || typeof f.content === 'string');
      if (!hasContent) return;
      out.push({
        field,
        filename: f.filename || 'attachment',
        type: f.contentType || 'application/octet-stream',
        // Convert Buffer → base64 for SendGrid
        content: Buffer.isBuffer(f.content) ? f.content.toString('base64') : f.content,
      });
    };
    if (Array.isArray(val)) val.forEach(pushFile);
    else if (val && typeof val === 'object' && ('filename' in val || 'content' in val)) pushFile(val);
  }
  return out;
};

const buildHtml = (p) => {
  // Create a simple, readable HTML summary of key fields.
  const rows = [
    ['Business', p.businessName],
    ['Contact', p.contactName],
    ['Email', p.email],
    ['Phone', p.phone],
    ['Website', p.website],
    ['Tagline', p.tagline],
    ['Short Description', p.shortDescription],
    ['Pages Selected', p.pagesSelected],
    ['Features Selected', p.featuresSelected],
    ['Copywriting', p.copywriting],
    ['SEO', p.seo],
    ['Timeline', p.timeline],
    ['Package Interest', p.packageInterest],
  ];

  const safe = (s) => String(s ?? '').replace(/</g, '&lt;').replace(/>/g, '&gt;');

  return `
    <h2 style="margin:0 0 12px;font-family:Inter,Arial,sans-serif;">New Client Onboarding</h2>
    <table cellpadding="6" cellspacing="0" border="0" style="font-family:Inter,Arial,sans-serif; border-collapse:collapse;">
      ${rows.map(([k,v]) => `
        <tr>
          <td style="vertical-align:top;font-weight:600;">${safe(k)}:</td>
          <td style="vertical-align:top;">${safe(v)}</td>
        </tr>
      `).join('')}
    </table>
  `.trim();
};

export const handler = async (event) => {
  if (event.httpMethod !== 'POST') {
    return { statusCode: 405, body: 'Method Not Allowed' };
  }

  try {
    const rawCT = H(event.headers || {}, 'content-type') || '';
    const ct = rawCT.toLowerCase();

    let parsed = {};
    let files = [];

    // Parse multipart (Netlify bodies are base64 for multipart; parser handles it)
    if (ct.includes('multipart/form-data')) {
      parsed = multipart.parse(event, true) || {};
      files = extractFiles(parsed);
      // Remove file-like entries from fields before validation
      for (const k of Object.keys(parsed)) {
        const v = parsed[k];
        if (v && typeof v === 'object' && ('filename' in v || 'content' in v)) {
          delete parsed[k];
        }
      }
    } else if (ct.includes('application/x-www-form-urlencoded')) {
      const sp = new URLSearchParams(event.body || '');
      for (const [k, v] of sp.entries()) parsed[k] = v;
    } else if (ct.includes('application/json')) {
      parsed = JSON.parse(event.body || '{}');
    } else {
      // Fallback attempt (useful for local testing)
      try { parsed = JSON.parse(event.body || '{}'); } catch { parsed = {}; }
    }

    // Normalize expected fields
    const fields = {
      businessName:     str(parsed.businessName).trim(),
      contactName:      str(parsed.contactName).trim(),
      email:            str(parsed.email).trim(),
      phone:            str(parsed.phone).trim(),
      website:          str(parsed.website).trim(),
      tagline:          str(parsed.tagline).trim(),
      shortDescription: str(parsed.shortDescription).trim(),
      pagesSelected:    str(parsed.pagesSelected).trim(),
      featuresSelected: str(parsed.featuresSelected).trim(),
      copywriting:      str(parsed.copywriting).trim(),
      seo:              str(parsed.seo).trim(),
      timeline:         str(parsed.timeline).trim(),
      packageInterest:  str(parsed.packageInterest).trim(),
    };

    // Required fields
    if (!fields.businessName || !fields.contactName || !fields.email) {
      return {
        statusCode: 400,
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          ok: false,
          error: 'Missing required fields (businessName, contactName, email).',
          receivedFlags: {
            businessName: !!fields.businessName,
            contactName:  !!fields.contactName,
            email:        !!fields.email,
          },
        }),
      };
    }

    // Build and send email
    const toEmail = process.env.ONBOARDING_TO_EMAIL || 'team@launchin7.io';
    const fromEmail = process.env.FROM_EMAIL || 'Team <team@launchin7.io>';

    const html = buildHtml(fields);

    const msg = {
      to: toEmail,
      from: fromEmail, // must be a verified sender/domain in SendGrid
      subject: `Onboarding • ${fields.businessName}`,
      html,
      attachments: files.map((f) => ({
        content: f.content,        // base64
        filename: f.filename,
        type: f.type,
        disposition: 'attachment',
      })),
    };

    await sgMail.send(msg);

    return {
      statusCode: 200,
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ ok: true, message: 'Submission sent.' }),
    };
  } catch (err) {
    console.error('client-onboarding error:', err);
    return {
      statusCode: 500,
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ ok: false, error: 'Server error while processing form.' }),
    };
  }
};
